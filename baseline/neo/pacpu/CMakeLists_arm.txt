cmake_minimum_required(VERSION 3.18 FATAL_ERROR)
project(PagedAttentionCPU_ARM C CXX)

# --------------------------------------------------------------------------
# ARM/aarch64 build — replaces the ISPC-based CMakeLists.txt.
# Uses OpenMP SIMD + GCC auto-vectorization (NEON/SVE2) instead of ISPC.
# Tested on GH200 (Neoverse V2, ARMv9.2-A) with GCC 11, PyTorch 2.7.
# --------------------------------------------------------------------------

set(TARGET_NAME_PREFIX pacpu)
set(TARGET_SOURCES "pacpu.cpp")   # pacpu_arm.h included transitively via core.h

find_package(Torch REQUIRED)

set(GEN_TORCH_LIBRARY TRUE)

function(target_add_arm_options target model_name tp_degree)
  set_property(TARGET ${target} PROPERTY CXX_STANDARD 17)
  set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)

  # -O3 + -march=native enables SVE2/NEON auto-vectorization on Neoverse V2.
  # __ARM_FP16_FORMAT_IEEE is set automatically by -march=native on aarch64.
  target_compile_options(${target} PRIVATE
    -O3
    -march=native
    -Ofast
    -funroll-loops
    -ffast-math
    -fopenmp           # enables #pragma omp simd
  )

  target_link_options(${target} PRIVATE -fopenmp)

  find_package(OpenMP REQUIRED)
  target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)

  string(TOUPPER ${model_name} MODEL_NAME)
  string(TOUPPER ${tp_degree}  TP_DEGREE)
  target_compile_definitions(${target} PRIVATE "${MODEL_NAME}" TP_DEGREE=${TP_DEGREE})
endfunction()

function(gen_torch_lib model_name tp_degree)
  set(TARGET_NAME "${TARGET_NAME_PREFIX}-${model_name}-tp${tp_degree}")
  add_library(${TARGET_NAME} SHARED)
  target_sources(${TARGET_NAME} PRIVATE ${TARGET_SOURCES})
  target_add_arm_options(${TARGET_NAME} ${model_name} ${tp_degree})
  target_compile_options(${TARGET_NAME} PRIVATE ${TORCH_CXX_FLAGS})
  target_link_libraries(${TARGET_NAME} PRIVATE "${TORCH_LIBRARIES}")
  target_include_directories(${TARGET_NAME} PRIVATE ${TORCH_INCLUDE_DIRS})
endfunction()

if(GEN_TORCH_LIBRARY)
  gen_torch_lib(${Model} ${TP})
endif()
